import 'package:flutter/material.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:app_divisor_de_cuentas/main.dart';

void main() {
  testWidgets('Flujo real: 100% calcula 50.00 y 150% retira el resultado', (tester) async {
    await tester.pumpWidget(crearAplicacion());
    await tester.enterText(find.byKey(const Key('monto')), '100');
    await tester.enterText(find.byKey(const Key('personas')), '4');
    await tester.enterText(find.byKey(const Key('propina')), '100');
    await tester.tap(find.text('Calcular'));
    await tester.pumpAndSettle();
    expect(find.text('50.00'), findsOneWidget);
    expect(find.byKey(const Key('resultado')), findsOneWidget);
    await tester.enterText(find.byKey(const Key('propina')), '150');
    await tester.tap(find.text('Calcular'));
    await tester.pumpAndSettle();
    expect(find.text('La propina no puede superar el 100%'), findsOneWidget);
    expect(find.byKey(const Key('resultado')), findsNothing);
    expect(find.text('50.00'), findsNothing);
    await tester.enterText(find.byKey(const Key('propina')), '100');
    await tester.tap(find.text('Calcular'));
    await tester.pumpAndSettle();
    expect(find.text('50.00'), findsOneWidget);
    expect(find.text('La propina no puede superar el 100%'), findsNothing);
  });
}
